\begin{figure*}[t]
  \centering
  \includegraphics[width=\textwidth]{\tilearenafigdir/main/fig_rq1_cross_accelerator.pdf}
  \caption{Proximity to modeled SOL, $T_{\text{SOL}}/T_k$, of each tile DSL on each accelerator.
  $T_{\text{SOL}}=\max(F/P_{\text{peak}},\,Q/BW_{\text{peak}})$ uses the operator's frozen operation count $F$ and memory
  traffic $Q$ and the peak of the arithmetic mode that the algorithm and its
  numerical contract require: the vendor's published dense rate for the direct GEMM and the empirically measured
  sustained rate otherwise, a hybrid SOL reference (memory-only for copies, indexing, comparisons and integer
  operators); $T_k$ is the formal
  autotuned latency. Each cell is the geometric mean over the operators of a workload category (row; \emph{Overall}:
  all 45 operators, split below into 13 memory-only and 32 compute+memory targets) of each operator's geometric mean
  over its valid input cases. Each column is relative to its own device's hybrid SOL reference; values do not compare
  absolute latency or hardware capability. MI300X has only Triton results.}
  \label{fig:rq1-cross-accelerator}
\end{figure*}
